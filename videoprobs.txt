          </div>

      <script src="https://www.youtube.com/iframe_api"></script>
      <script>
        let player = null;
        let currentVideoId = null;
        let activeVideos = [];

        const cfgEl = document.getElementById('free-quiz-config');
        const THRESHOLD = parseInt(cfgEl.dataset.threshold, 10);
        const VIDEOS = JSON.parse(cfgEl.dataset.videos || '[]');

        function pickRandom(arr) {
          if (!arr || !arr.length) return null;
          return arr[Math.floor(Math.random() * arr.length)];
        }

        function normalizeKeywords(raw) {
          if (!raw) return [];
          return String(raw)
            .toLowerCase()
            .split(/[^a-z0-9]+/g)
            .map(s => s.trim())
            .filter(Boolean);
        }

        function filterBySearch(query) {
          const kws = normalizeKeywords(query);
          if (!kws.length) return VIDEOS;

          return VIDEOS.filter(v => {
            const vk = (v.keywords || []).map(x => String(x).toLowerCase());
            const help = String(v.help || '').toLowerCase();
            return kws.some(k => vk.includes(k) || vk.some(x => x.includes(k)) || help.includes(k));
          });
        }

        function renderTiles(arr) {
          const root = document.getElementById('video-tiles');
          root.innerHTML = '';

          if (!arr.length) {
            root.innerHTML = '<div style="padding:10px; color:#666;">No related videos found.</div>';
            return;
          }

          arr.forEach(v => {
            const tile = document.createElement('button');
            tile.type = 'button';
            tile.className = 'free-video-tile';
            tile.dataset.videoId = v.id;
            tile.innerHTML = `
              <div class="free-video-tile-top">Video</div>
              <div class="free-video-tile-preview">${v.id ? `<iframe class="free-video-tile-iframe" src="https://www.youtube.com/embed/${v.id}?autoplay=0&mute=1&controls=0&modestbranding=1&rel=0&playsinline=1&enablejsapi=0" frameborder="0" loading="lazy" referrerpolicy="origin" allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture; web-share" allowfullscreen></iframe>` : ''}</div>
              <div class="free-video-tile-text">${String(v.help || '').replace(/[<>]/g,'')}</div>
            `;
            tile.addEventListener('click', () => {
              selectVideo(v.id);
            });
            root.appendChild(tile);
          });

          highlightSelected(currentVideoId);
        }

        function highlightSelected(videoId) {
          document.querySelectorAll('.free-video-tile').forEach(btn => {
            btn.classList.toggle('active', btn.dataset.videoId === videoId);
          });
        }

        function selectVideo(videoId) {
          currentVideoId = videoId;
          highlightSelected(videoId);
          if (!player || !videoId) return;
          try {
            player.loadVideoById(videoId);
          } catch (e) {
            // ignore
          }
        }

        function renderForCurrentSearch(autoPick = true) {
          const searchEl = document.getElementById('video-search');
          activeVideos = filterBySearch(searchEl ? searchEl.value : '');
          renderTiles(activeVideos);

          if (autoPick) {
            const picked = pickRandom(activeVideos);
            if (picked) selectVideo(picked.id);
          }
        }

        window.addEventListener('load', () => {
          const searchEl = document.getElementById('video-search');
          const searchBtn = document.getElementById('video-search-btn');
          const otherBtn = document.getElementById('other-videos-btn');

          if (searchEl) {
            searchEl.addEventListener('input', () => {
              renderForCurrentSearch(false);
            });
          }
          if (searchBtn) {
            searchBtn.addEventListener('click', () => {
              renderForCurrentSearch(true);
            });
          }
          if (otherBtn) {
            otherBtn.addEventListener('click', () => {
              renderForCurrentSearch(true);
            });
          }

          renderForCurrentSearch(true);
        });

        window.onYouTubeIframeAPIReady = function () {
          if (!activeVideos || !activeVideos.length) activeVideos = VIDEOS;

          const initialPick = pickRandom(activeVideos);
          currentVideoId = initialPick ? initialPick.id : (VIDEOS[0] ? VIDEOS[0].id : '{{ video_id }}');

          player = new YT.Player('player', {
            videoId: currentVideoId,
            playerVars: {
              autoplay: 0,
              mute: 1,

              controls: 1,
              modestbranding: 1,
              rel: 0
            },
            events: {
              onReady: () => {
                try { player.pauseVideo(); } catch (e) {}
                highlightSelected(currentVideoId);
              }
            }
          });

          let lastAwardedBucket = -1;
          setInterval(() => {
            if (!player) return;

            const time = Math.floor(player.getCurrentTime());
            const bucket = Math.floor(time / THRESHOLD);

            const inBucket = time % THRESHOLD;
            const timeDisp = document.getElementById('time-display');
            if (timeDisp) timeDisp.innerText = (time < THRESHOLD) ? time : inBucket;

            if (bucket >= 1 && bucket > lastAwardedBucket) {
              lastAwardedBucket = bucket;

              const secondsForRequest = bucket * THRESHOLD;
              fetch('/update-progress', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({
                  seconds_watched: secondsForRequest,
                  bucket_index: bucket
                })
              })
                .then(res => res.json())
                .then(data => {
                  const creditsEl = document.getElementById('credits-count');
                  if (creditsEl) creditsEl.innerText = data.credits;
                })
                .catch(() => {
                  lastAwardedBucket = bucket - 1;
                });
            }
          }, 1000);
        };
      </script>
